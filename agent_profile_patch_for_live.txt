PROFILE_FILE = "agent_profiles.json"
AGENT_PROFILES = {}

def load_agent_profiles():
    global AGENT_PROFILES
    p = Path(PROFILE_FILE)
    if not p.exists():
        AGENT_PROFILES = {}
        print("[PROFILE] agent_profiles.json not found, using defaults.")
        return
    data = json.loads(p.read_text(encoding="utf-8"))
    AGENT_PROFILES = data.get("profiles", {})
    print(f"[PROFILE] Loaded {len(AGENT_PROFILES)} trained profiles.")

def profile_for(symbol: str):
    prof = AGENT_PROFILES.get(symbol)
    if not prof or not prof.get("enabled", False):
        return None
    return prof

def trained_param(symbol: str, key: str, default):
    prof = profile_for(symbol)
    if not prof:
        return default
    return prof.get("params", {}).get(key, default)
